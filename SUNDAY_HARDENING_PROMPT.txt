CAMBIO 5 — Puppeteer en Worker Thread  (estimado: 3-4 horas)
----------------------------------------------------------------

Crear backend/src/pdfWorker.ts como worker thread dedicado:

    import { workerData, parentPort } from 'worker_threads';
    // Toda la lógica de Puppeteer se mueve aquí
    // Al terminar:
    parentPort.postMessage({ success: true, buffer });
    // En error:
    parentPort.postMessage({ success: false, error: err.message });

En htmlPdfGenerator.ts, reemplazar la llamada directa por:

    import { Worker } from 'worker_threads';

    export function generatePdfFromDocument(params): Promise<Buffer> {
      return new Promise((resolve, reject) => {
        const worker = new Worker('./dist/pdfWorker.js', {
          workerData: params
        });
        worker.on('message', (result) => {
          if (result.success) resolve(result.buffer);
          else reject(new Error(result.error));
        });
        worker.on('error', reject);
        // Timeout de seguridad: 90 segundos
        setTimeout(() => {
          worker.terminate();
          reject(new Error('PDF generation timeout'));
        }, 90000);
      });
    }

NOTA: El worker necesita acceso a las mismas variables de
entorno. Pasarlas via workerData o usar dotenv en el worker.

CAMBIO 6 — Build de Producción con Nginx  (estimado: 2 horas)
----------------------------------------------------------------

Crear archivo nginx.conf en la raíz del proyecto:

    worker_processes 1;
    events { worker_connections 1024; }

    http {
      include mime.types;
      default_type application/octet-stream;
      proxy_read_timeout 120s;
      proxy_connect_timeout 10s;

      server {
        listen 5173;
        server_name _;
        root /ruta/al/frontend/dist;
        index index.html;

        # SPA routing
        location / {
          try_files $uri $uri/ /index.html;
        }

        # API proxy
        location /api/ {
          proxy_pass http://localhost:3001;
          proxy_http_version 1.1;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # Socket.IO WebSocket
        location /socket.io/ {
          proxy_pass http://localhost:3001;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection "upgrade";
          proxy_set_header Host $host;
          proxy_read_timeout 86400s;
        }

        # Archivos subidos
        location /uploads/ {
          proxy_pass http://localhost:3001;
        }
      }
    }

Actualizar .env.example del frontend:

    # Reemplazar X.X con la IP real del servidor
    VITE_API_URL=http://192.168.X.X:3001
